            System.out.println("ERROR: MONGODB_URI environment variable is not set.");

            companies = db.getCollection("companies");
            jobDrives = db.getCollection("jobDrives");

            System.out.println("Connected to MongoDB Atlas. Database: college");
            System.out.println("Collections used: students, companies, jobDrives, applications");

            mainMenu();
        } catch (Exception e) {
            System.out.println("Application/database error: " + e.getMessage());
            System.out.println("Check MONGODB_URI, Atlas network access, credentials, and database permissions.");
        }
    }

    private static void addStudent() {
        String id = readRequired("Student ID (unique): ");
        if (students.find(Filters.eq("studentId", id)).first() != null) {
            System.out.println("A student with that ID already exists.");
            return;
        }

        String name = readRequired("Full name: ");
        String department = readRequired("Department (e.g. CSE): ");
        double cgpa = readDouble("CGPA (0-10): ");
        if (cgpa < 0 || cgpa > 10) {
            System.out.println("CGPA must be between 0 and 10.");
            return;
        }
        String email = readRequired("Email: ");
        String phone = readRequired("Phone: ");
        List<String> skills = readCsvList("Skills (comma-separated; blank for none): ");

        Document doc = new Document("studentId", id)
                .append("name", name)
                .append("department", department)
                .append("cgpa", cgpa)
                .append("email", email)
                .append("phone", phone)
                .append("skills", skills)
                .append("createdAt", Instant.now().toString());

        students.insertOne(doc);
        System.out.println("Student added. MongoDB _id: " + doc.getObjectId("_id"));
    }

    private static void updateStudent() {
        String id = readRequired("Student ID to update: ");
        Document existing = students.find(Filters.eq("studentId", id)).first();
        if (existing == null) {
            System.out.println("Student not found.");
            return;
        }

        System.out.println("Leave a field blank to keep its current value.");
        String name = readLine("Name [" + existing.getString("name") + "]: ").trim();
        String dept = readLine("Department [" + existing.getString("department") + "]: ").trim();
        String email = readLine("Email [" + existing.getString("email") + "]: ").trim();
        String phone = readLine("Phone [" + existing.getString("phone") + "]: ").trim();
        String cgpaText = readLine("CGPA [" + existing.get("cgpa") + "]: ").trim();

        List<org.bson.conversions.Bson> changes = new ArrayList<>();
        if (!name.isEmpty()) changes.add(Updates.set("name", name));
        if (!dept.isEmpty()) changes.add(Updates.set("department", dept));
        if (!email.isEmpty()) changes.add(Updates.set("email", email));
        if (!phone.isEmpty()) changes.add(Updates.set("phone", phone));
        if (!cgpaText.isEmpty()) {
            try {
                double cgpa = Double.parseDouble(cgpaText);
                if (cgpa < 0 || cgpa > 10) {
                    System.out.println("CGPA must be between 0 and 10. No update made.");
                    return;
                }
                changes.add(Updates.set("cgpa", cgpa));
            } catch (NumberFormatException e) {
                System.out.println("Invalid CGPA. No update made.");
                return;
            }
        }

        String skillsText = readLine("Replace skills with comma-separated values (blank to keep): ").trim();
        if (!skillsText.isEmpty()) changes.add(Updates.set("skills", parseCsv(skillsText)));

        if (changes.isEmpty()) {
            System.out.println("No changes entered.");
            return;
        }
        students.updateOne(Filters.eq("studentId", id), Updates.combine(changes));
        System.out.println("Student updated.");
    }

    private static void deleteStudent() {
        String id = readRequired("Student ID to delete: ");
        long related = applications.countDocuments(Filters.eq("studentId", id));
        if (related > 0) {
            System.out.println("Cannot delete: " + related + " application(s) reference this student.");
            System.out.println("Withdraw/delete those applications first.");
            return;
        }
        long deleted = students.deleteOne(Filters.eq("studentId", id)).getDeletedCount();
        System.out.println(deleted == 1 ? "Student deleted." : "Student not found.");
    }

    private static void addDrive() {
        String id = readRequired("Drive ID (unique): ");

            return;
        }
        String companyId = readRequired("Company ID: ");
        Document company = companies.find(Filters.eq("companyId", companyId)).first();
        if (company == null) {
            System.out.println("Company ID does not exist. Register the company first.");
            return;
        }

        double minCgpa = readDouble("Minimum CGPA (0-10): ");
        if (minCgpa < 0 || minCgpa > 10) {
            System.out.println("Minimum CGPA must be between 0 and 10.");
            return;
        }
        double salary = readDouble("Annual salary/package (numeric amount): ");
        if (salary < 0) {

        Document doc = new Document("driveId", id)
                .append("companyId", companyId)

                .append("minCgpa", minCgpa)

                .append("createdAt", Instant.now().toString());
        jobDrives.insertOne(doc);
        System.out.println("Job drive created.");
    }

    private static void updateDrive() {
        String id = readRequired("Drive ID to update: ");
        Document existing = jobDrives.find(Filters.eq("driveId", id)).first();
        if (existing == null) {
            System.out.println("Job drive not found.");
            return;
        }
        System.out.println("Leave a field blank to keep its current value.");
        List<org.bson.conversions.Bson> changes = new ArrayList<>();
        String role = readLine("Role [" + existing.getString("role") + "]: ").trim();
        String dept = readLine("Eligible department [" + existing.getString("eligibleDepartment") + "]: ").trim();
        String minText = readLine("Minimum CGPA [" + existing.get("minCgpa") + "]: ").trim();
        String salaryText = readLine("Salary [" + existing.get("salary") + "]: ").trim();
        String deadline = readLine("Deadline [" + existing.getString("deadline") + "]: ").trim();
        String status = readLine("Status (Open/Closed) [" + existing.getString("status") + "]: ").trim();

        if (!role.isEmpty()) changes.add(Updates.set("role", role));
        if (!dept.isEmpty()) changes.add(Updates.set("eligibleDepartment", dept));
        if (!deadline.isEmpty()) changes.add(Updates.set("deadline", deadline));
        if (!status.isEmpty()) {
            if (!status.equalsIgnoreCase("Open") && !status.equalsIgnoreCase("Closed")) {
                System.out.println("Status must be Open or Closed. No update made.");
                return;
            }
            changes.add(Updates.set("status", status.substring(0, 1).toUpperCase() + status.substring(1).toLowerCase()));
        }
        try {
            if (!minText.isEmpty()) {
                double value = Double.parseDouble(minText);
                if (value < 0 || value > 10) throw new IllegalArgumentException();
                changes.add(Updates.set("minCgpa", value));
            }
            if (!salaryText.isEmpty()) {
                double value = Double.parseDouble(salaryText);
                if (value < 0) throw new IllegalArgumentException();
                changes.add(Updates.set("salary", value));
            }
        } catch (Exception e) {
            System.out.println("Invalid CGPA or salary. No update made.");
            return;
        }
        if (changes.isEmpty()) {
            System.out.println("No changes entered.");
            return;
        }
        jobDrives.updateOne(Filters.eq("driveId", id), Updates.combine(changes));
        System.out.println("Job drive updated.");
    }

    private static void deleteDrive() {

        }
        long deleted = jobDrives.deleteOne(Filters.eq("driveId", id)).getDeletedCount();
        System.out.println(deleted == 1 ? "Job drive deleted." : "Job drive not found.");
    }

    private static void applyToDrive() {
        String studentId = readRequired("Student ID: ");
        Document student = students.find(Filters.eq("studentId", studentId)).first();
        if (student == null) {
            System.out.println("Student not found. Add the student first.");
            return;
        }

        String driveId = readRequired("Drive ID: ");
        Document drive = jobDrives.find(Filters.eq("driveId", driveId)).first();
        if (drive == null) {
            System.out.println("Job drive not found.");
            return;
        }
        if (!"Open".equalsIgnoreCase(drive.getString("status"))) {
            System.out.println("This job drive is closed.");
            return;
        }

        String department = student.getString("department");
        String eligibleDepartment = drive.getString("eligibleDepartment");
        double cgpa = numberValue(student.get("cgpa"), -1);
        double minCgpa = numberValue(drive.get("minCgpa"), 0);

        if (eligibleDepartment != null && !"ALL".equalsIgnoreCase(eligibleDepartment)
                && !eligibleDepartment.equalsIgnoreCase(department)) {
            System.out.println("Student department does not meet the eligibility criteria.");
            return;
        }
        if (cgpa < minCgpa) {
            System.out.println("Student CGPA does not meet the minimum requirement.");
            return;
        }
        if (applications.find(Filters.and(
                Filters.eq("studentId", studentId),
                Filters.eq("driveId", driveId))).first() != null) {
            System.out.println("This student has already applied to this drive.");
            return;
        }

        Document application = new Document("applicationId", "APP-" + new ObjectId().toHexString())
                .append("studentId", studentId)
                .append("studentName", student.getString("name"))
                .append("driveId", driveId)
                .append("companyId", drive.getString("companyId"))
                .append("companyName", drive.getString("companyName"))
                .append("role", drive.getString("role"))
                .append("status", "Applied")
                .append("appliedAt", Instant.now().toString());
        applications.insertOne(application);
        System.out.println("Application submitted. Application ID: " + application.getString("applicationId"));
    }

    private static void updateApplicationStatus() {
        String id = readRequired("Application ID: ");
        Document existing = applications.find(Filters.eq("applicationId", id)).first();
        if (existing == null) {
            System.out.println("Application not found.");
            return;
        }
        System.out.println("Allowed statuses: Applied, Shortlisted, Interview, Selected, Rejected, Withdrawn");
        String status = readRequired("New status: ");
        String normalized = normalizeApplicationStatus(status);
        if (normalized == null) {
            System.out.println("Invalid status.");
            return;
        }
        applications.updateOne(Filters.eq("applicationId", id),
                Updates.combine(Updates.set("status", normalized),
                        Updates.set("updatedAt", Instant.now().toString())));
        System.out.println("Application status updated.");
    }

    private static void withdrawApplication() {
        String id = readRequired("Application ID to withdraw: ");
        long updated = applications.updateOne(Filters.eq("applicationId", id),
                Updates.combine(Updates.set("status", "Withdrawn"),
                        Updates.set("updatedAt", Instant.now().toString()))).getModifiedCount();
        System.out.println(updated == 1 ? "Application withdrawn." : "Application not found or already withdrawn.");
    }

    // -------------------- CANDIDATE SEARCH --------------------

    private static void candidateSearch() {
        System.out.println("\n--- CANDIDATE SEARCH ---");
        System.out.println("Leave any filter blank to skip it.");
        String department = readLine("Department: ").trim();
        String minText = readLine("Minimum CGPA: ").trim();
        String skill = readLine("Required skill: ").trim();

        List<org.bson.conversions.Bson> filters = new ArrayList<>();
        if (!department.isEmpty()) filters.add(Filters.regex("department", "^" + regexQuote(department) + "$", "i"));
        if (!skill.isEmpty()) filters.add(Filters.regex("skills", "^" + regexQuote(skill) + "$", "i"));
        if (!minText.isEmpty()) {
            try {
                double min = Double.parseDouble(minText);
                if (min < 0 || min > 10) {
                    System.out.println("Minimum CGPA must be between 0 and 10.");
                    return;
                }
                filters.add(Filters.gte("cgpa", min));
            } catch (NumberFormatException e) {
                System.out.println("Invalid CGPA.");
                return;
            }
        }

        org.bson.conversions.Bson filter = filters.isEmpty() ? new Document() : Filters.and(filters);
        printDocuments(students.find(filter).sort(Sorts.descending("cgpa")));
    }

    private static void departmentStats() {
        System.out.println("\n--- DEPARTMENT-WISE STUDENT STATISTICS ---");
        for (Document row : students.aggregate(List.of(
                Aggregates.match(Filters.exists("department", true)),
                Aggregates.group("$department",
                        Accumulators.sum("studentCount", 1),
                        Accumulators.avg("averageCgpa", "$cgpa")),
                Aggregates.sort(Sorts.ascending("_id"))))) {
            System.out.println("Department: " + row.get("_id")
                    + " | Students: " + row.get("studentCount")
                    + " | Average CGPA: " + formatNumber(row.get("averageCgpa")));
        }
    }

    private static void averageCgpaByDepartment() {
        System.out.println("\n--- AVERAGE CGPA BY DEPARTMENT ---");
        for (Document row : students.aggregate(List.of(
                Aggregates.match(Filters.and(
                        Filters.exists("department", true),
                        Filters.exists("cgpa", true))),
                Aggregates.group("$department",
                        Accumulators.avg("averageCgpa", "$cgpa"),
                        Accumulators.sum("studentCount", 1)),
                Aggregates.sort(Sorts.descending("averageCgpa"))))) {
            System.out.println("Department: " + row.get("_id")
                    + " | Average CGPA: " + formatNumber(row.get("averageCgpa"))
                    + " | Students counted: " + row.get("studentCount"));
        }
    }

    private static void topStudents() {
        int limit = readInt("How many top students? ");
        if (limit < 1 || limit > 100) {
            System.out.println("Enter a number from 1 to 100.");
            return;
        }
        printDocuments(students.find(Filters.exists("cgpa", true))
                .sort(Sorts.descending("cgpa")).limit(limit));
    }

    private static void applicationStatusStats() {
        System.out.println("\n--- APPLICATION STATUS STATISTICS ---");
        for (Document row : applications.aggregate(List.of(
                Aggregates.group("$status", Accumulators.sum("count", 1)),
                Aggregates.sort(Sorts.ascending("_id"))))) {
            System.out.println("Status: " + row.get("_id") + " | Count: " + row.get("count"));
        }
    }

    private static void placementSummary() {
        long totalStudents = students.countDocuments();
        long totalApplications = applications.countDocuments();
        long selectedApplications = applications.countDocuments(Filters.eq("status", "Selected"));
        long placedStudents = applications.distinct("studentId", String.class)
                .into(new ArrayList<>()).stream()
                .filter(id -> applications.find(Filters.and(
                        Filters.eq("studentId", id),
                        Filters.eq("status", "Selected"))).first() != null)
                .count();

        double rate = totalStudents == 0 ? 0.0 : (placedStudents * 100.0 / totalStudents);
        System.out.println("\n--- PLACEMENT SUMMARY ---");
        System.out.println("Registered students: " + totalStudents);
        System.out.println("Total applications: " + totalApplications);
        System.out.println("Selected applications: " + selectedApplications);
        System.out.println("Unique students with Selected status: " + placedStudents);
        System.out.printf("Student placement rate: %.2f%%%n", rate);
        System.out.println("Note: this is based on applications whose status is exactly 'Selected'.");
    }

    // -------------------- COMMON HELPERS --------------------

    private static void printDocuments(Iterable<Document> docs) {
        boolean found = false;
        for (Document doc : docs) {
            System.out.println("----------------------------------------");
            System.out.println(doc.toJson());
            found = true;
        }
        if (!found) System.out.println("No matching records found.");
        else System.out.println("----------------------------------------");
    }

    private static void printOne(Document doc) {
        if (doc == null) System.out.println("Record not found.");
        else System.out.println(doc.toJson());
    }

    private static String readLine(String prompt) {
        System.out.print(prompt);
        return SC.nextLine();
    }

    private static String readRequired(String prompt) {
        while (true) {
            String value = readLine(prompt).trim();
            if (!value.isEmpty()) return value;
            System.out.println("This field cannot be empty.");
        }
    }

    private static int readInt(String prompt) {
        while (true) {
            String value = readLine(prompt).trim();
            try {
                return Integer.parseInt(value);
            } catch (NumberFormatException e) {
                System.out.println("Enter a valid whole number.");
            }
        }
    }

    private static double readDouble(String prompt) {
        while (true) {
            String value = readLine(prompt).trim();
            try {
                return Double.parseDouble(value);
            } catch (NumberFormatException e) {
                System.out.println("Enter a valid number.");
            }
        }
    }

    private static List<String> readCsvList(String prompt) {
        return parseCsv(readLine(prompt));
    }

    private static List<String> parseCsv(String value) {
        List<String> result = new ArrayList<>();
        if (value == null || value.isBlank()) return result;
        for (String item : value.split(",")) {
            String trimmed = item.trim();
            if (!trimmed.isEmpty() && !result.contains(trimmed)) result.add(trimmed);
        }
        return result;
    }

    private static String normalizeApplicationStatus(String value) {
        String[] allowed = {"Applied", "Shortlisted", "Interview", "Selected", "Rejected", "Withdrawn"};
        for (String status : allowed) {
            if (status.equalsIgnoreCase(value.trim())) return status;
        }
        return null;
    }

    private static double numberValue(Object value, double fallback) {
        if (value instanceof Number n) return n.doubleValue();
        return fallback;
    }

    private static String formatNumber(Object value) {
        if (!(value instanceof Number n)) return "N/A";
        return String.format("%.2f", n.doubleValue());
    }

    private static String regexQuote(String value) {
        return java.util.regex.Pattern.quote(value);
    }
